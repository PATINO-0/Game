# Exporta mallas seleccionadas y coordenadas compatibles con Three.js.






import bpy # [doc] Accede a escenas y operadores de Blender.
import os # [doc] Permite crear carpetas y construir rutas.
import json # [doc] Permite escribir los registros de posición.
from mathutils import Vector # [doc] Importa Vector; el script actual no lo utiliza.


nivel = 1 # [doc] Selecciona el nivel que quedará en los nombres y registros.


export_path = r"D:/UCC/React/PortafolioUCC/Projecto_final_v1/game-project/public/models/toycar2" # [doc] Define la carpeta de salida; debe adaptarse a esta copia del repositorio.
json_path = os.path.join(export_path, f"toy_car_blocks{nivel}.json") # [doc] Construye el nombre del JSON de posiciones del nivel.
os.makedirs(export_path, exist_ok=True) # [doc] Crea la carpeta de salida si todavía no existe.

positions = [] # [doc] Inicia el arreglo de registros exportados.


depsgraph = bpy.context.evaluated_depsgraph_get() # [doc] Obtiene el grafo evaluado de la escena.
bpy.ops.object.mode_set(mode='OBJECT') # [doc] Cambia a modo Objeto para permitir la selección.
bpy.context.view_layer.update() # [doc] Actualiza las transformaciones de la escena.


for obj in bpy.context.selected_objects: # [doc] Recorre la lista de objetos seleccionados al iniciar el bucle.
    if obj.type != 'MESH': # [doc] Permite exportar únicamente objetos de malla.
        continue # [doc] Omite el objeto que no es una malla.

    obj_eval = obj.evaluated_get(depsgraph) # [doc] Lee el objeto evaluado con sus transformaciones finales.
    matrix_world = obj_eval.matrix_world # [doc] Obtiene su matriz de transformación mundial.
    loc = matrix_world.to_translation() # [doc] Extrae la posición mundial del objeto.


    name = obj.name.lower().replace(" ", "_").replace(f"_lev{nivel}", "") # [doc] Normaliza el nombre y retira el sufijo del nivel actual.


    name_with_level = f"{name}_lev{nivel}" # [doc] Añade el sufijo del nivel a exportar.
    obj.name = name_with_level # [doc] Actualiza el nombre de la malla en la sesión de Blender.


    pos = { # [doc] Construye el registro que consumirá el juego.
        "name": obj.name, # [doc] Conserva la clave que debe coincidir con sources.js.
        "x": loc.x, # [doc] Conserva el eje X de Blender.
        "y": loc.z, # [doc] Convierte la altura Z de Blender en Y.
        "z": -loc.y, # [doc] Convierte Y de Blender en Z con signo invertido.
        "level": nivel, # [doc] Asocia el registro al nivel exportado.
        "role": "default" # [doc] Asigna el rol normal; una moneda final necesita otro valor.
    }

    positions.append(pos) # [doc] Añade el registro a la lista de salida.
    print(f"📦 {obj.name}: ({pos['x']}, {pos['y']}, {pos['z']}, nivel {pos['level']})") # # [doc] Informa el avance o la finalización en la consola de Blender.


    bpy.ops.object.select_all(action='DESELECT') # [doc] Limpia la selección antes de exportar una sola malla.
    obj.select_set(True) # [doc] Selecciona la malla actual.
    bpy.context.view_layer.objects.active = obj # [doc] Define la malla activa para el exportador.

    filepath = os.path.join(export_path, f"{obj.name}.glb") # [doc] Construye la ruta del GLB individual.
    bpy.ops.export_scene.gltf( # [doc] Exporta la selección con el operador glTF de Blender.
        filepath=filepath, # [doc] Indica el archivo de salida del modelo.
        use_selection=True, # [doc] Limita la exportación al objeto seleccionado.
        export_format='GLB', # [doc] Selecciona el contenedor binario GLB.
        export_apply=False # [doc] Conserva el estado actual de aplicación de modificadores.
    )


with open(json_path, "w") as f: # [doc] Abre el archivo de posiciones para escritura.
    json.dump(positions, f, indent=4) # [doc] Serializa todos los registros con sangría legible.

print("✅ Exportación finalizada con sistema de coordenadas compatible con Three.js.") # [ # [doc] Informa el avance o la finalización en la consola de Blender.
